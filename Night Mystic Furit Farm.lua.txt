local function Bv(F)
	if not F.Character or not F.Character:FindFirstChild("HumanoidRootPart") then
		return false;
	end;
	if F.Character:FindFirstChild("InSafeZone") and F.Character.InSafeZone.Value then
		return true;
	end;
	if F.Character:FindFirstChildOfClass("ForceField") then
		return true;
	end;
	local n = workspace:FindFirstChild("_WorldOrigin") and workspace._WorldOrigin:FindFirstChild("SafeZones");
	if n then
		for n, B in pairs(n:GetChildren()) do
			if not B:IsA("BasePart") then
				continue;
			end;
			local Y = B.CFrame:PointToObjectSpace(F.Character.HumanoidRootPart.Position);
			local V = math.abs(Y.X) <= B.Size.X / 2;
			if V then
				V = math.abs(Y.Y) <= B.Size.Y / 2 and math.abs(Y.Z) <= B.Size.Z / 2;
			end;
			if V then
				return true;
			end;
		end;
	end;
	return false;
end;
